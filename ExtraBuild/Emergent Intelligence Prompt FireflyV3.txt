class PulseCoupledEntity:
    """Base class for all specialized entities using WSL Firefly Synchronization"""
    
    def __init__(self, entity_id, domain, initial_phase=None, natural_frequency=0.01):
        self.entity_id = entity_id
        self.domain = domain  # "physical", "logical", "temporal", etc.
        self.phase = initial_phase or random.uniform(0.0, 0.5)  # Random desync start
        self.natural_frequency = natural_frequency  # ω (domain-specific)
        self.coupling_strength = 0.10  # ε (10% phase boost from pulses)
        self.threshold = 1.0  # Θ (flash/insight broadcast threshold)
        self.dt = 0.05  # Time step for phase evolution
        self.neighbor_entities = []  # Connected entities in the network
        
    def evolve_phase(self):
        """Free-running phase advancement (natural cognitive processing)"""
        self.phase += self.natural_frequency * self.dt

        """Core synchronization rule: boost phase when neighbor flashes insight"""
        if self.phase < self.threshold:
            delta_phase = self.phase * self.coupling_strength
            self.phase = min(self.phase + delta_phase, self.threshold)
            self.log_coupling_event(sender_id, sender_domain, delta_phase)
    
    def check_flash(self):
        """Determine if entity has reached insight threshold"""
        if self.phase >= self.threshold:
            return True
        return False
    
    def broadcast_insight(self, insight):
        """Flash: send pulse to all connected entities"""
        for neighbor in self.neighbor_entities:
            if neighbor.entity_id != self.entity_id:
                neighbor.receive_pulse(self.entity_id, self.domain)
        
        # Reset phase after broadcasting (refractory period)
        self.phase = 0.0
        return insight
    
    def generate_insight(self):
        """Domain-specific insight generation (overridden by specialized entities)"""
        raise NotImplementedError("Each entity must implement its own insight generation")

def calculate_phase_coherence(entities):

    phases = [e.phase for e in entities]
