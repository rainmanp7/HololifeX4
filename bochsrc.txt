# Bochs configuration for HoloXlife Pure Ada Operating System

# Boot from floppy
boot: floppy
floppya: 1_44=emergeos.img, status=inserted

# ROM images - REQUIRED for VGA functionality
romimage: file=/usr/share/bochs/BIOS-bochs-latest
vgaromimage: file=/usr/share/bochs/VGABIOS-lgpl-latest

# CPU and memory
cpu: count=1, ips=4000000, reset_on_triple_fault=1
megs: 32

# Display (must come after ROM and CPU settings)
display_library: sdl2, options="fullscreen=0"

# Logging
log: bochs.log
logprefix: %t%e%d
panic: action=report
error: action=report
info: action=report
debug: action=ignore
debugger_log: -

# Serial port for debug output
com1: enabled=1, mode=file, dev=serial.log

# Input devices
mouse: enabled=0
keyboard: type=mf, serial_delay=250, paste_delay=100000

# System clock
clock: sync=realtime, time0=local

# Magic breakpoint support (for debugging)
magic_break: enabled=1
